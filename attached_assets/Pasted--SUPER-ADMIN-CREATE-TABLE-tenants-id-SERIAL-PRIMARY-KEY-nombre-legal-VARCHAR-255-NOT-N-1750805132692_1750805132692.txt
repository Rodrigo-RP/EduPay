-- SUPER ADMIN
CREATE TABLE tenants (
    id SERIAL PRIMARY KEY,
    nombre_legal VARCHAR(255) NOT NULL,
    rfc VARCHAR(13) NOT NULL,
    cfdi_pac_id VARCHAR(255),
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- CAMPUS DE ESCUELAS
CREATE TABLE campuses (
    id SERIAL PRIMARY KEY,
    tenant_id INTEGER REFERENCES tenants(id) ON DELETE CASCADE,
    nombre VARCHAR(255) NOT NULL,
    clave_sep VARCHAR(50),
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- USUARIOS (Admin, Caja, Contador)
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    campus_id INTEGER REFERENCES campuses(id) ON DELETE CASCADE,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(50) CHECK (role IN ('super_admin', 'admin', 'caja', 'contador')),
    twofa_secret VARCHAR(255),
    is_active BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- ALUMNOS
CREATE TABLE students (
    id SERIAL PRIMARY KEY,
    campus_id INTEGER REFERENCES campuses(id) ON DELETE CASCADE,
    curp VARCHAR(18),
    nombre_completo VARCHAR(255) NOT NULL,
    grado VARCHAR(50),
    grupo VARCHAR(50),
    status VARCHAR(50) CHECK (status IN ('activo', 'baja', 'suspendido', 'egresado')) DEFAULT 'activo',
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- RESPONSABLES DE PAGO (Padres, Empresas)
CREATE TABLE guardians (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) NOT NULL,
    telefono VARCHAR(20),
    nombre_completo VARCHAR(255) NOT NULL,
    rfc VARCHAR(13),
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- RELACION ALUMNO → RESPONSABLE
CREATE TABLE student_guardian (
    student_id INTEGER REFERENCES students(id) ON DELETE CASCADE,
    guardian_id INTEGER REFERENCES guardians(id) ON DELETE CASCADE,
    porcentaje_responsabilidad NUMERIC(5,2) DEFAULT 100.00,
    PRIMARY KEY (student_id, guardian_id)
);

-- CONCEPTOS DE PAGO
CREATE TABLE concepts (
    id SERIAL PRIMARY KEY,
    campus_id INTEGER REFERENCES campuses(id) ON DELETE CASCADE,
    nombre VARCHAR(255) NOT NULL,
    tipo VARCHAR(50) CHECK (tipo IN ('colegiatura', 'inscripcion', 'extra')),
    periodicidad VARCHAR(50) CHECK (periodicidad IN ('mensual', 'anual', 'eventual')),
    monto_centavos BIGINT NOT NULL,
    iva BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- BECAS INDIVIDUALES
CREATE TABLE scholarships (
    id SERIAL PRIMARY KEY,
    student_id INTEGER REFERENCES students(id) ON DELETE CASCADE,
    porcentaje NUMERIC(5,2) NOT NULL,
    vigencia_inicio DATE NOT NULL,
    vigencia_fin DATE NOT NULL,
    motivo VARCHAR(255),
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- CARGOS EMITIDOS
CREATE TABLE charges (
    id SERIAL PRIMARY KEY,
    student_id INTEGER REFERENCES students(id) ON DELETE CASCADE,
    concept_id INTEGER REFERENCES concepts(id),
    ciclo_escolar VARCHAR(50),
    fecha_emision DATE NOT NULL,
    fecha_vencimiento DATE NOT NULL,
    monto_base_centavos BIGINT NOT NULL,
    beca_aplicada NUMERIC(5,2) DEFAULT 0.00,
    recargo_aplicado_centavos BIGINT DEFAULT 0,
    estado VARCHAR(50) CHECK (estado IN ('pendiente', 'pagado', 'parcial', 'cancelado')) DEFAULT 'pendiente',
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- PAGOS REGISTRADOS
CREATE TABLE payments (
    id SERIAL PRIMARY KEY,
    charge_id INTEGER REFERENCES charges(id) ON DELETE CASCADE,
    guardian_id INTEGER REFERENCES guardians(id),
    metodo VARCHAR(50) CHECK (metodo IN ('tarjeta', 'spei', 'paypal', 'efectivo', 'oxxo')),
    referencia_pasarela VARCHAR(255),
    monto_centavos BIGINT NOT NULL,
    fecha_pago TIMESTAMP DEFAULT now(),
    estado VARCHAR(50) CHECK (estado IN ('exitoso', 'fallido', 'reversado')) DEFAULT 'exitoso',
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- METODOS DE PAGO TOKENIZADOS
CREATE TABLE payment_methods (
    id SERIAL PRIMARY KEY,
    guardian_id INTEGER REFERENCES guardians(id) ON DELETE CASCADE,
    tipo VARCHAR(50) CHECK (tipo IN ('card', 'spei', 'paypal')),
    token_pasarela VARCHAR(255),
    last4 VARCHAR(4),
    expiry VARCHAR(10),
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- FACTURAS CFDI
CREATE TABLE invoices (
    id SERIAL PRIMARY KEY,
    payment_id INTEGER REFERENCES payments(id) ON DELETE CASCADE,
    uuid_cfdi VARCHAR(255),
    xml_url TEXT,
    pdf_url TEXT,
    estado VARCHAR(50) CHECK (estado IN ('emitido', 'cancelado')) DEFAULT 'emitido',
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- DESCUENTOS POR PRONTO PAGO
CREATE TABLE discounts (
    id SERIAL PRIMARY KEY,
    campus_id INTEGER REFERENCES campuses(id) ON DELETE CASCADE,
    nombre VARCHAR(255),
    regla_sql TEXT,
    monto_pct NUMERIC(5,2),
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);

-- NOTIFICACIONES ENVIADAS
CREATE TABLE notifications (
    id SERIAL PRIMARY KEY,
    user_id INTEGER REFERENCES users(id),
    canal VARCHAR(50) CHECK (canal IN ('email', 'sms', 'whatsapp')),
    contenido TEXT,
    enviado_en TIMESTAMP DEFAULT now()
);

-- CONCILIACION BANCARIA
CREATE TABLE reconciliation_batches (
    id SERIAL PRIMARY KEY,
    campus_id INTEGER REFERENCES campuses(id),
    banco VARCHAR(255),
    fecha_inicial DATE NOT NULL,
    fecha_final DATE NOT NULL,
    archivo_csv TEXT,
    estado VARCHAR(50) CHECK (estado IN ('pendiente', 'conciliado', 'error')) DEFAULT 'pendiente',
    created_at TIMESTAMP DEFAULT now(),
    updated_at TIMESTAMP DEFAULT now()
);
