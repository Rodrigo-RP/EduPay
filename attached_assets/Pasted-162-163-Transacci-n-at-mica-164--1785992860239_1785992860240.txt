162: 
163:       // ── Transacción atómica ──────────────────────────────────────────────
164:       const client = await pool.connect();
165:       let paymentId!: number;
166:       let montoAplicado!: number;
167:       let newEstado!: string;
168:       try {
169:         await client.query("BEGIN");
170: 
171:         // Lock del cargo — serializa doble cobro concurrente
172:         const lockRes = await client.query(
173:           `SELECT id, monto_base_centavos, estado
174:            FROM charges WHERE id = $1 AND tenant_id = $2 FOR UPDATE`,
175:           [chargeId, tenantIdCaja]
176:         );
177:         if (!(lockRes.rows as any[]).length) {
178:           await client.query("ROLLBACK");
179:           return res.json({ message: "Cargo no encontrado", payment_id: null, monto_centavos: montoOperador });
180:         }
181:         const locked = (lockRes.rows as any[])[0];
182: 
183:         if (["pagado", "cancelado"].includes(locked.estado)) {
184:           await client.query("ROLLBACK");
185:           return res.json({ message: "El cargo ya fue pagado o cancelado", payment_id: null, monto_centavos: montoOperador });
186:         }
187: 
188:         // Saldo pendiente real (dentro del mismo client)
189:         const saldoRes = await client.query(
190:           `SELECT COALESCE(SUM(pa.amount_centavos), 0)::bigint AS ya_pagado
191:            FROM payment_applications pa WHERE pa.charge_id = $1`,
192:           [chargeId]
193:         );
194:         const yaPagado = Number((saldoRes.rows as any[])[0].ya_pagado);
195:         const saldoPendiente = Number(locked.monto_base_centavos) - yaPagado;
196: 
197:         if (saldoPendiente <= 0) {
198:           await client.query("ROLLBACK");
199:           return res.json({ message: "El cargo ya tiene saldo cero", payment_id: null, monto_centavos: montoOperador });
200:         }
201: 
202:         // Calcular cuánto se aplica al cargo y cuánto sobra
203:         montoAplicado = Math.min(montoOperador, saldoPendiente);
204:         const excedente = montoOperador - montoAplicado; // 0 si no sobra nada
205:         newEstado = montoAplicado >= saldoPendiente ? "pagado" : "parcial";
206: 
207:         const referencia = `CAJA-${Date.now()}`;
208:         // El payment registra el MONTO TOTAL COBRADO (efectivo recibido en caja)
209:         // — importante para cuadre de caja. La payment_application aplica solo
210:         // lo que cubre el cargo.
211:         const payRow = await client.query(
212:           `INSERT INTO payments
213:              (tenant_id, charge_id, guardian_id, metodo, referencia_pasarela,
214:               monto_centavos, fecha_pago, estado)
215:            VALUES ($1,$2,NULL,'efectivo',$3,$4,CURRENT_DATE,'exitoso') RETURNING id`,
216:           [tenantIdCaja, chargeId, referencia, montoOperador]
217:         );
218:         paymentId = (payRow.rows as any[])[0].id;
219: 
220:         // Ledger entry: solo la parte que cubre el cargo
221:         await client.query(
222:           `INSERT INTO payment_applications (payment_id, charge_id, amount_centavos, applied_at)
223:            VALUES ($1,$2,$3,NOW())`,
224:           [paymentId, chargeId, montoAplicado]
225:         );
226: 
227:         await client.query(
228:           `UPDATE charges SET estado = $1, updated_at = NOW() WHERE id = $2`,
229:           [newEstado, chargeId]
230:         );
231: 
232:         // Si hubo excedente → registrar como saldo a favor de la familia
233:         if (excedente > 0) {
234:           const familyRow = await client.query(
235:             `SELECT family_id FROM family_students WHERE student_id = $1 LIMIT 1`,
236:             [estudiante_id]
237:           );
238:           const creditFamilyId = (familyRow.rows as any[])[0]?.family_id ?? null;
239: 
240:           await client.query(
241:             `INSERT INTO family_credits
242:                (tenant_id, campus_id, family_id, student_id, payment_id,
243:                 amount_centavos, origen, descripcion)
244:              VALUES ($1,$2,$3,$4,$5,$6,'excedente_caja',$7)`,
245:             [
246:               tenantIdCaja, campusId, creditFamilyId, estudiante_id, paymentId,
247:               excedente,
248:               `Cambio en cobro de caja — pago $${(montoOperador / 100).toFixed(2)}, cargo $${(saldoPendiente / 100).toFixed(2)}`,
249:             ]
250:           );
251:         }
252: 
253:         await client.query("COMMIT");
254:       } catch (err) {
255:         await client.query("ROLLBACK");
256:         throw err;
257:       } finally {
258:         client.release();
259:       }
260: 
261:       // ── Audit fuera de la transacción (ADR-001) ──────────────────────────
262:       pool.query(
263:         `INSERT INTO audit_log
264:            (tenant_id, user_id, action, entity_type, entity_id, new_value, metadata)
265:          VALUES ($1,$2,'charge.status_changed','charge',$3,$4,$5)`,
266:         [
267:           tenantIdCaja, userIdCaja, chargeId,
268:           JSON.stringify({ estado: newEstado }),
269:           JSON.stringify({
270:             flujo: "caja_efectivo", payment_id: paymentId,
271:             monto_operador: montoOperador, monto_aplicado: montoAplicado,
272:             recibido_por, observaciones,
273:           }),
274:         ]
275:       ).catch(() => {});
276: 
277:       const excedente = montoOperador - montoAplicado;
278:       res.json({
279:         message: `Pago en efectivo registrado (${newEstado})${excedente > 0 ? ` — $${(excedente / 100).toFixed(2)} de cambio registrado como saldo a favor` : ""}`,
280:         payment_id:              paymentId,
281:         monto_aplicado_centavos: montoAplicado,
282:         monto_centavos:          montoOperador,
283:         excedente_centavos:      excedente,
284:         charge_nuevo_estado:     newEstado,
285:       });
286:     } catch (error: any) {
287:       res.status(500).json({ message: "Error interno del servidor" });
288:     }
289:   });
