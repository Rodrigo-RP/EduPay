import { useState } from "react";
import { Card, CardHeader, CardTitle, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { BarChart3, CreditCard, Users2, ShieldCheck } from "lucide-react";

// Simple mock‑ups for the 5 key screens of the Payments SaaS
// TailwindCSS + shadcn/ui components

const navItems = [
  "Onboarding Wizard",
  "Admin Dashboard",
  "Concept Config",
  "Parent Dashboard",
  "Checkout Modal",
] as const;

type Screen = (typeof navItems)[number];

export default function MockupsPaymentPlatform() {
  const [screen, setScreen] = useState<Screen>("Admin Dashboard");

  return (
    <div className="w-full min-h-screen bg-gray-100 p-4">
      {/* Navigation */}
      <div className="flex gap-2 mb-4 flex-wrap">
        {navItems.map((item) => (
          <Button
            key={item}
            variant={screen === item ? "default" : "outline"}
            onClick={() => setScreen(item)}
            className="text-sm"
          >
            {item}
          </Button>
        ))}
      </div>

      {/* Screen Renderer */}
      <div className="bg-white p-6 rounded-2xl shadow w-full overflow-auto">
        {screen === "Onboarding Wizard" && <OnboardingWizard />}
        {screen === "Admin Dashboard" && <AdminDashboard />}
        {screen === "Concept Config" && <ConceptConfig />}
        {screen === "Parent Dashboard" && <ParentDashboard />}
        {screen === "Checkout Modal" && <CheckoutModal />}
      </div>
    </div>
  );
}
